    --outdir ./eda_out


2. 데이터 전처리 (샘플로 바로 dry-run 가능)
bash
cd deepretrieval_mind/code/data_preprocess
python mind_feedback.py \
    --behaviors ../../sample_data/behaviors.tsv \
    --news ../../sample_data/news.tsv \
    --outdir ../../sample_out \
    --max_history 10 --min_candidates 2 --test_ratio 0.25
